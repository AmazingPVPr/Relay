cmake_minimum_required(VERSION 3.24)
project(RelayWindowsLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core CorePrivate Network Widgets)
find_package(OpenSSL REQUIRED)
add_executable(relay-launcher updater.cpp)
target_link_libraries(relay-launcher PRIVATE Qt6::Core Qt6::CorePrivate Qt6::Network Qt6::Widgets OpenSSL::Crypto)
set_source_files_properties(../../../updater/release-key.pem PROPERTIES QT_RESOURCE_ALIAS updater/release-key.pem)
qt_add_resources(relay-launcher release_key PREFIX / FILES ../../../updater/release-key.pem)
if(WIN32)
 enable_language(RC)
 target_sources(relay-launcher PRIVATE ../../../assets/relay.rc)
 target_include_directories(relay-launcher PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../assets")
 set_target_properties(relay-launcher PROPERTIES WIN32_EXECUTABLE TRUE)
 add_executable(Relay WIN32 bootstrap.cpp ../../../assets/relay.rc)
 target_include_directories(Relay PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../assets")
 target_compile_definitions(Relay PRIVATE UNICODE _UNICODE)
 target_link_libraries(Relay PRIVATE shell32)
 if(MINGW)
  target_link_options(Relay PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
endif()

include(CTest)
if(BUILD_TESTING)
 add_executable(windows_updater_test test.cpp)
 target_link_libraries(windows_updater_test PRIVATE Qt6::Core Qt6::CorePrivate Qt6::Network Qt6::Widgets OpenSSL::Crypto)
 qt_add_resources(windows_updater_test test_key PREFIX / FILES ../../../updater/release-key.pem)
 add_test(NAME windows_updater_test COMMAND windows_updater_test)
endif()
install(TARGETS relay-launcher RUNTIME DESTINATION bin)
if(WIN32)
 install(TARGETS Relay RUNTIME DESTINATION .)
endif()
